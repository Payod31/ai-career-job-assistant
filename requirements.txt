# Python backend dependencies for the AI endpoints.
# Install with:  .venv\Scripts\python -m pip install -r requirements.txt
# (or, from a fresh venv:  python -m venv .venv first.)
#
# Versions are PINNED to the exact set the automated test suite
# (tests/test_backend.py) was run against, so a Render deploy (`pip install
# -r requirements.txt`) rebuilds the same environment the tests verified.
# Unpinned `>=` ranges let a fresh build drift to untested versions — e.g.
# a new flask-limiter/limits combo running in the request path that the
# local suite never exercised.

flask==3.1.3
werkzeug==3.1.9
openai==3.37.0
python-dotenv==1.2.4
# Server-side resume text extraction for uploaded files (see doc_extract.py):
pypdf==6.21.2
python-docx==1.2.0
# Production WSGI server for Render (local dev can still use `python server.py`):
gunicorn>=21.0
# Per-client rate limiting for the expensive AI routes (in-memory by default;
# set RATELIMIT_STORAGE_URI for shared storage). Not applied to static files.
flask-limiter==4.1.1
# flask-limiter's runtime engine, pinned to the version the tests ran against
# (already a transitive dependency of flask-limiter — not a new library):
limits==5.8.0
